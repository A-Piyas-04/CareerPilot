\rowcolor{rowalt}
\textbf{Hugging Face} &
Optional model downloads for local embeddings &
\envvar{HF\_TOKEN} (transformers backend only) \\

\sectiontitle{6. Install Commands}
% ===========================================================================

\begin{codeblock}
# Frontend
cd frontend && npm install

# Backend
cd backend && pip install -r requirements.txt

# Root (Supabase CLI)
npm install

# Full stack (Docker)
docker compose up --build
\end{codeblock}
